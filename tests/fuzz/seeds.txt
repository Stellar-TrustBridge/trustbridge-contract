# Cursor fuzz regression seeds (Issue #650): known-bad cursors derived from
# tests/cursor_pagination.rs edge cases.  Each u64 encodes a full 8-byte
# cursor (offset u32 BE | generation u32 BE) followed implicitly by limit=0.
#
# Cursor layout reminder (src/storage.rs encode_cursor):
#   bytes 0..4 = offset  (big-endian u32)
#   bytes 4..8 = generation (big-endian u32)
#
# Stale-generation cursors (generation != 0, any offset):

0x00000002FFFFFFFF
